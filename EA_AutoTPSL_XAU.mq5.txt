#property strict

input double TP_Point = 500;   // TP dalam point. 500 point = 5.0 di broker 3 digit
input double SL_Point = 500;   // SL dalam point
input bool   OnlyOnePosition = false; // UDAH DIGANTI: false = Boleh banyak posisi

void OnTradeTransaction(const MqlTradeTransaction& trans,
                        const MqlTradeRequest& request,
                        const MqlTradeResult& result)
{
   // Jalanin tiap ada order baru keisi
   if(trans.type == TRADE_TRANSACTION_DEAL_ADD)
   {
      CheckAndModify();
   }
}

void OnInit()
{
   Print("EA Auto TP/SL XAU 500 Point Aktif - Mode Multi Posisi");
}

void CheckAndModify()
{
   for(int i=0; i<PositionsTotal(); i++)
   {
      ulong ticket = PositionGetTicket(i);
      if(ticket > 0)
      {
         string symbol = PositionGetString(POSITION_SYMBOL);
         double tp = PositionGetDouble(POSITION_TP);
         double sl = PositionGetDouble(POSITION_SL);
         long magic = PositionGetInteger(POSITION_MAGIC);
         
         // Cuma proses XAUUSD dan order manual magic=0 dan belum ada TP/SL
         if(symbol == _Symbol && magic == 0 && tp == 0 && sl == 0)
         {
            double price = PositionGetDouble(POSITION_PRICE_OPEN);
            int type = (int)PositionGetInteger(POSITION_TYPE);
            double point = _Point;
            
            double new_tp, new_sl;
            
            MqlTradeRequest req;
            MqlTradeResult res;
            ZeroMemory(req);
            
            req.action = TRADE_ACTION_SLTP;
            req.position = ticket;
            req.symbol = _Symbol;
            
            if(type == POSITION_TYPE_BUY)
            {
               new_tp = price + TP_Point * point;
               new_sl = price - SL_Point * point;
            }
            else // SELL
            {
               new_tp = price - TP_Point * point;
               new_sl = price + SL_Point * point;
            }
            
            req.tp = NormalizeDouble(new_tp, _Digits);
            req.sl = NormalizeDouble(new_sl, _Digits);
            
            // Cek return value
            if(!OrderSend(req, res))
            {
               Print("Gagal pasang TP/SL. Error: ", res.retcode, " Ticket: ", ticket);
            }
            else
            {
               Print("Sukses pasang TP/SL. Ticket: ", ticket);
            }
            
            // Pengaman 1 posisi UDAH DIMATIIN
            // if(OnlyOnePosition) CloseOtherPositions(ticket);  <-- ini udah gak jalan
         }
      }
   }
}
